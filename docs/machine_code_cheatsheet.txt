# A NOTE ON FORMATTING

x.n     n denotes how many bits x uses
u = unsigned
s = signed
r = register
[....] = bits

# REGISTERS
r0-15
zero = r0
fp   = r13
sp   = r14
lr   = r15

# PSEUDO-COMMANDS

LI rd, imm32        LHI rd, imm32[31:16]
                    LLI rd, imm32[15:0]

LA rd, label        LI rd, address(label)

B label             JMP simm26(label)

CALL label          JAL simm26(label)

.byte x             emit x.8

.word x             emit x.32 big-endian

.asciz "text"       emit text bytes followed by 00.8

.org addr           set/pad output address

NOTE

Branch/jump offsets are relative to the PC value at execute time. In the current emulator, PC has already advanced past the fetched instruction before execute.
